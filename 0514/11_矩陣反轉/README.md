問題描述 ：  
請設計一程式，輸入一個陣列並且反轉後再輸出。  
輸入說明 ：  
第一行先輸入矩陣的行、列，之後再輸入陣列元素。  
輸出說明 ：  
反轉後的矩陣。  
範例 ：  

| 輸入範例 | 輸出範例 |
|---|---|
| 2 1                                                  <br>3                                                      <br>2                                                      <br>3 2                                                            4 6                                                              7 8                                                                   4 6                                                                 2 4                                                                     7 8 9 10                                                                   6 5 22 4                                                             4 2                                                                       4 5                                                                     2 3                                                                 4 5                                                                 6 9 | 3 2                                                                   4 7 4                                                                  6 8 6                                                                7 6                                                                     8 5                                                                    9 22                                                                 10 4                                                                      4 2 4 6                                                                 5 3 5 9 |

> 💡 **筆記**  
>
> 回圈內宣告 int d[c][r];//產生陣列，直接存成轉置矩陣  
> 輸入時cin >> d[j][i];  //原本應該存入 d[i][j]，但陣列 d 的結構是 轉置後的 c × r，  
> d[j][i] 直接把行列對調，讓輸入的時候就變成轉置矩陣！  
> if (j == r - 1) //如果是最後一個數字則直接輸出  

```cpp
#include<iostream>

using namespace std;

int main()
{
	int r,c;

	while(cin >> r >> c)//輸入陣列大小
	{
        int d[c][r];//產生陣列
        for (int i = 0; i < r; i++) // 讀入次數即列數
        {
            for (int j = 0; j < c; j++) // 字串分割數即行數
            {
                cin >> d[j][i]; // 一一讀入指定的反矩陣座標
            }
        }

        // 輸出反矩陣座標
        for (int i = 0; i < c; i++)
        {
            for (int j = 0; j < r; j++)
            {
                if (j == r - 1)
                {
                    cout << d[i][j] << endl;
                }
                else
                {
                    cout << d[i][j] << " ";
                }
            }
        }
	}

	return 0;
}
